function fmtBytes(b: number) {
  if (!b) return '0 B'
  const units = ['B','KB','MB','GB','TB']
  let i = 0; let n = b
  while (n >= 1024 && i < units.length - 1) { n /= 1024; i++ }
  return `${n.toFixed(i > 0 ? 1 : 0)} ${units[i]}`
}

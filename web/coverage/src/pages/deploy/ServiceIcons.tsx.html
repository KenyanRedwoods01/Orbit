export function getServiceIcon(name: string, repo?: string): string {
  const lower = (name + ' ' + (repo || '')).toLowerCase()
  if (lower.includes('github')) return 'github'
  if (lower.includes('gitlab')) return 'gitlab'
  if (lower.includes('bitbucket')) return 'bitbucket'
  if (lower.includes('gitea')) return 'gitea'
  if (lower.includes('redis')) return 'redis'
  if (lower.includes('postgres') || lower.includes('pg') || lower.includes('psql')) return 'postgres'
  if (lower.includes('mysql') || lower.includes('mariadb')) return 'mysql'
  if (lower.includes('mongo')) return 'mongodb'
  if (lower.includes('nginx')) return 'nginx'
  if (lower.includes('node') || lower.includes('npm') || lower.includes('yarn')) return 'nodejs'
  if (lower.includes('docker')) return 'docker'
  return 'default'
}
